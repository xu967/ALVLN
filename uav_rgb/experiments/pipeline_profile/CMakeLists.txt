cmake_minimum_required(VERSION 3.20)
set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "CUDA architectures" FORCE)
project(uavrgb_pipeline_profile LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CUDAToolkit REQUIRED)
add_subdirectory(../../src core)
add_executable(pipeline_profile pipeline_profile.cpp)
target_link_libraries(pipeline_profile PRIVATE uavrgb_core CUDA::cudart)
target_compile_options(pipeline_profile PRIVATE -Wall -Wextra)
